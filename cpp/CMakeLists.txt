cmake_minimum_required(VERSION 3.16)
project(zora_coins VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ZORA_BUILD_TESTS "Build the tests" ${PROJECT_IS_TOP_LEVEL})

include(FetchContent)
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
  FetchContent_MakeAvailable(json)
endif()
find_package(CURL REQUIRED)

add_library(zora_coins src/curl_transport.cpp)
add_library(zora::coins ALIAS zora_coins)
target_include_directories(zora_coins PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(zora_coins PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl)
target_compile_features(zora_coins PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(zora_coins PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(zora-rewards examples/zora_rewards.cpp)
target_link_libraries(zora-rewards PRIVATE zora::coins)

if(ZORA_BUILD_TESTS)
  enable_testing()
  add_executable(zora_tests tests/tests.cpp)
  target_link_libraries(zora_tests PRIVATE zora::coins)
  target_compile_definitions(zora_tests PRIVATE ZORA_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../fixtures")
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(zora_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME zora_tests COMMAND zora_tests)
endif()

install(TARGETS zora_coins EXPORT zora_coins_targets)
install(DIRECTORY include/ DESTINATION include)
